cmake_minimum_required(VERSION 3.21)
project(CurvedPoisson LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CURVED_POISSON_NEKTAR "Build the upstream Nektar++ adapter" OFF)
find_path(FFTW3_INCLUDE_DIR fftw3.h REQUIRED)
find_library(FFTW3_LIBRARY fftw3 REQUIRED)
foreach(target poisson_single poisson_nodal)
  if(target STREQUAL "poisson_single")
    set(driver single.cpp)
  else()
    set(driver nodal_single.cpp)
  endif()
  add_executable(${target} benchmarks/nonconformal_bspf_opt/${driver})
  target_compile_options(${target} PRIVATE -O3)
  target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/nonconformal_bspf_opt)
endforeach()
add_executable(fd4 benchmarks/finite_difference_poisson/solver.cpp)
target_compile_options(fd4 PRIVATE -O3)
target_include_directories(fd4 PRIVATE ${FFTW3_INCLUDE_DIR})
target_link_libraries(fd4 PRIVATE ${FFTW3_LIBRARY} ${CMAKE_DL_LIBS})
set_target_properties(fd4 PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/finite_difference_poisson)
if(CURVED_POISSON_NEKTAR)
  find_package(Nektar++ REQUIRED CONFIG)
  add_executable(nektar_reference benchmarks/standard_poisson/nektar.cpp)
  target_compile_options(nektar_reference PRIVATE -O3)
  target_compile_definitions(nektar_reference PRIVATE ${NEKTAR++_DEFINITIONS})
  target_include_directories(nektar_reference PRIVATE ${NEKTAR++_INCLUDE_DIRS} ${NEKTAR++_TP_INCLUDE_DIRS})
  target_link_directories(nektar_reference PRIVATE ${NEKTAR++_LIBRARY_DIRS} ${NEKTAR++_TP_LIBRARY_DIRS})
  foreach(component LibUtilities LocalRegions SpatialDomains StdRegions Collections MultiRegions MatrixFreeOps)
    find_library(NEKTAR_${component} NAMES ${component} PATHS ${NEKTAR++_LIBRARY_DIRS} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(nektar_reference PRIVATE ${NEKTAR_${component}})
  endforeach()
  target_link_libraries(nektar_reference PRIVATE ${NEKTAR++_TP_LIBRARIES} ${CMAKE_DL_LIBS})
  set_target_properties(nektar_reference PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/standard_poisson/build)
endif()
