cmake_minimum_required(VERSION 3.21)
project(BSPFPoisson3D LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
option(BSPF_POISSON_DIRECT_LAYOUT "Experiment: permutation-free tensor layout (not faster on RTX 3070)" OFF)
find_package(CUDAToolkit REQUIRED)
add_library(bspf_cuda_poisson STATIC ../../../src/cuda/poisson3d.cu)
target_include_directories(bspf_cuda_poisson PUBLIC ../../../include)
if(BSPF_POISSON_DIRECT_LAYOUT)
  target_compile_definitions(bspf_cuda_poisson PRIVATE BSPF_POISSON_DIRECT_LAYOUT)
endif()
target_link_libraries(bspf_cuda_poisson PUBLIC CUDA::cudart CUDA::cublas)
add_executable(bspf_poisson3d solver.cpp)
target_link_libraries(bspf_poisson3d PRIVATE bspf_cuda_poisson ${CMAKE_DL_LIBS})
