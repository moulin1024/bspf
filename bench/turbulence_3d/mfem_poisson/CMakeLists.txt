cmake_minimum_required(VERSION 3.21)
project(curved_poisson3d_mfem_ceed LANGUAGES CXX)
find_package(MFEM 4.9 CONFIG REQUIRED)
if(NOT MFEM_USE_CEED)
  message(FATAL_ERROR "MFEM must be built with MFEM_USE_CEED=ON")
endif()
if(NOT MFEM_USE_DOUBLE)
  message(FATAL_ERROR "This benchmark requires double-precision MFEM")
endif()
add_executable(mfem_poisson3d solver.cpp)
target_compile_features(mfem_poisson3d PRIVATE cxx_std_17)
target_link_libraries(mfem_poisson3d PRIVATE mfem ${CMAKE_DL_LIBS})
if(MFEM_USE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(NOT CMAKE_CUDA_COMPILER)
    set(CMAKE_CUDA_COMPILER "${CUDAToolkit_NVCC_EXECUTABLE}" CACHE FILEPATH "CUDA compiler")
  endif()
  enable_language(CUDA)
  # MFEM's SparseMatrix layout includes CUDA members only under __CUDACC__.
  # Match the CUDA-built MFEM ABI when constructing local CSR wrappers.
  set_source_files_properties(solver.cpp PROPERTIES LANGUAGE CUDA)
  target_sources(mfem_poisson3d PRIVATE lor_qinterp.cu)
  target_compile_options(mfem_poisson3d PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-extended-lambda;--expt-relaxed-constexpr>)
  set_target_properties(mfem_poisson3d PROPERTIES CUDA_ARCHITECTURES 86 CUDA_STANDARD 17)
  target_link_libraries(mfem_poisson3d PRIVATE CUDA::cudart)
  set(BSPF_HYPRE_FP32_ROOT "" CACHE PATH "Private static FP32 HYPRE installation")
  if(BSPF_HYPRE_FP32_ROOT)
    find_package(MPI REQUIRED COMPONENTS CXX)
    add_library(bspf_hypre_fp32 SHARED hypre_fp32_bridge.cu)
    target_include_directories(bspf_hypre_fp32 PRIVATE "${BSPF_HYPRE_FP32_ROOT}/include")
    target_link_libraries(bspf_hypre_fp32 PRIVATE
      "${BSPF_HYPRE_FP32_ROOT}/lib/libHYPRE.a" MPI::MPI_CXX
      CUDA::cudart CUDA::cusparse CUDA::curand CUDA::cublas CUDA::cublasLt CUDA::cusolver)
    set_target_properties(bspf_hypre_fp32 PROPERTIES CUDA_ARCHITECTURES 86 CUDA_STANDARD 17)
    target_link_options(bspf_hypre_fp32 PRIVATE
      "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/hypre_fp32.exports" "-Wl,-Bsymbolic")
    target_compile_definitions(mfem_poisson3d PRIVATE BSPF_HYPRE_FP32)
    target_link_libraries(mfem_poisson3d PRIVATE bspf_hypre_fp32)
  endif()
endif()
