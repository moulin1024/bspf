cmake_minimum_required(VERSION 3.21)
project(PETScFD8 LANGUAGES CXX CUDA)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PETSC REQUIRED PETSc)
find_package(CUDAToolkit REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(petsc_fd8 solver.cu)
set_target_properties(petsc_fd8 PROPERTIES CUDA_ARCHITECTURES 86 CUDA_STANDARD 17)
target_include_directories(petsc_fd8 PRIVATE ${PETSC_INCLUDE_DIRS})
target_link_directories(petsc_fd8 PRIVATE ${PETSC_STATIC_LIBRARY_DIRS})
target_link_libraries(petsc_fd8 PRIVATE ${PETSC_STATIC_LIBRARIES} MPI::MPI_CXX CUDA::cudart /tmp/bspf-mfem-poisson3d-fp32/libbspf_hypre_fp32.so)
target_link_options(petsc_fd8 PRIVATE ${PETSC_STATIC_LDFLAGS_OTHER})
